rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

function isTsmoaktAdmin() {
  return request.auth != null
    && request.auth.token.email == 'kevin@bloxit.be';
}

// Publiek kerstbestelformulier (kerst/): alleen aanmaken, nooit lezen of wijzigen.
function isValidPublicOrder() {
  let d = request.resource.data;
  return d.keys().hasOnly(['source', 'status', 'service', 'customerName', 'email', 'phone', 'adults', 'children',
      'location', 'eventDate', 'eventTime', 'pickupSlot', 'orderLines', 'orderTotal', 'calculatedTotal',
      'customerNotes', 'notes', 'formulaSelections', 'productSelections', 'createdAt', 'updatedAt'])
    && d.source == 'bestelformulier-kerst-2026'
    && d.status == 'todo'
    && d.customerName is string && d.customerName.size() >= 2 && d.customerName.size() <= 100
    && d.email is string && d.email.size() <= 120 && d.email.matches('^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$')
    && d.phone is string && d.phone.size() >= 6 && d.phone.size() <= 30
    && d.adults is int && d.adults >= 1 && d.adults <= 500
    && d.children == 0
    && d.eventDate in ['2026-12-24', '2026-12-25', '2026-12-26']
    && d.orderLines is list && d.orderLines.size() >= 1 && d.orderLines.size() <= 40
    && d.orderTotal is number && d.orderTotal > 0 && d.orderTotal < 20000
    && d.calculatedTotal == d.orderTotal
    && d.customerNotes is string && d.customerNotes.size() <= 1000
    && d.notes is string && d.notes.size() <= 5000
    && d.formulaSelections == [] && d.productSelections == []
    && d.createdAt == request.time && d.updatedAt == request.time;
}

match /tsmoakt_requests/{requestId} {
  allow read, create, update, delete: if isTsmoaktAdmin();
  allow create: if isValidPublicOrder();
  match /messages/{messageId} { allow read, write: if isTsmoaktAdmin(); }
  match /activities/{activityId} { allow read, write: if isTsmoaktAdmin(); }
  match /documents/{documentId} { allow read, write: if isTsmoaktAdmin(); }
}

match /tsmoakt_catalog/{documentId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_suppliers/{supplierId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_staff/{staffId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_products/{productId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_users/{userId} {
  allow read, write: if isTsmoaktAdmin();
}
  }
}
