rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

function isTsmoaktAdmin() {
  return request.auth != null
    && request.auth.token.email == 'kevin@bloxit.be';
}

match /tsmoakt_requests/{requestId} {
  allow read, create, update, delete: if isTsmoaktAdmin();
  match /messages/{messageId} { allow read, write: if isTsmoaktAdmin(); }
  match /activities/{activityId} { allow read, write: if isTsmoaktAdmin(); }
  match /documents/{documentId} { allow read, write: if isTsmoaktAdmin(); }
}

match /tsmoakt_catalog/{documentId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_suppliers/{supplierId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_staff/{staffId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_products/{productId} {
  allow read, write: if isTsmoaktAdmin();
}

match /tsmoakt_users/{userId} {
  allow read, write: if isTsmoaktAdmin();
}
  }
}
